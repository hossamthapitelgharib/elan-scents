TABLE private.guest_identity_blocks  rls=on  grants: anon=- authenticated=- service_role=-
  fingerprint_hash text NOT NULL
  analytics_hash text NOT NULL
  reason text NOT NULL
  blocked_at timestamp with time zone NOT NULL DEFAULT clock_timestamp()
  expires_at timestamp with time zone NOT NULL
  CONSTRAINT guest_identity_blocks_pkey PRIMARY KEY (fingerprint_hash)
  CONSTRAINT guest_identity_blocks_analytics_hash_key UNIQUE (analytics_hash)
  CONSTRAINT guest_identity_blocks_analytics_hash_check CHECK ((analytics_hash ~ '^[a-f0-9]{64}$'::text))
  CONSTRAINT guest_identity_blocks_check CHECK ((expires_at > blocked_at))
  CONSTRAINT guest_identity_blocks_fingerprint_hash_check CHECK ((fingerprint_hash ~ '^[a-f0-9]{64}$'::text))
  CONSTRAINT guest_identity_blocks_reason_check CHECK ((reason = ANY (ARRAY['refused'::text, 'claimed'::text])))
  CREATE INDEX guest_identity_blocks_expiry_idx ON private.guest_identity_blocks USING btree (expires_at)

  CREATE INDEX store_order_idempotency_request_idx ON private.store_order_idempotency USING btree (request_id)
